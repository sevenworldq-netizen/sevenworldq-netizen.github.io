//@version=5
// ============================================================================
// AMANDIRE S&R ZONE DETECTOR — v1.3 (colour settings)
// New vs v1.2 (Selam, 28 Nov 2026):
//   * Colour settings: the "Colours" group in Settings now lets you change
//     the resistance / support / flipped / violated / HTF box colours and
//     the zone label text colour. Boxes keep their fill transparency.
//     (Event labels BOUNCE/BREAK/RETEST keep fixed meaning-colours.)
// New vs v1.1 (MTF-only view):
//   * MTF-only view: with "MTF zones only" ON (default), a chart-TF zone is
//     DRAWN only where the CHECKED timeframes agree with it (confluence ≥
//     minimum: 3 = chart + 1 HTF, 5 = chart + 2 HTFs, 7 = all 3 HTFs).
//     Zones without agreement are hidden — no box, no label, no event
//     label, no alert — but the engine still tracks them, so if an HTF zone
//     later overlaps one, it appears. Turn it OFF to see all zones (v1.1).
//   * One checkbox per timeframe: chart TF here, HTF 1/2/3 in the "Higher
//     timeframes" group (their timeframes are inputs — set any you want).
//     Uncheck a timeframe and it no longer counts toward agreement.
//   * "Minimum confluence" input (1/3/5/7). HTF overlay boxes unchanged.
// Fixes vs v1.0 (kept):
//   1. `for` loops guarded by size > 0 (empty array crash).
//   2. `str` renamed to `sc` (shadowed the str.* namespace).
//   3. array.push(..., na) on int arrays now uses int(na).
//   4. HTF requests ignored when the HTF is not higher than the chart TF.
//   5. too-close filter no longer lets a later zone overwrite "suppress".
//
// NON-REPAINTING CONTRACT (unchanged)
//   * Zones are born only from confirmed pivots; frozen once drawn.
//   * Events run on closed bars only (barstate.isconfirmed).
//   * HTF via request.security(..., lookahead_off).
//   * A break = candle BODY close past the far edge; wicks are pierces.
//   * Touch is NEVER an entry by itself. Respect is not a trading edge.
// ============================================================================

indicator("Amandire S&R Zone Detector", shorttitle="Amandire SR Zones", overlay=true, max_boxes_count=500, max_labels_count=500, max_lines_count=500)

// ---------------- Inputs ----------------
grpZ  = "Zones"
grpH  = "Higher timeframes (MTF)"
grpD  = "Display"
grpA  = "Alerts"

pivLen     = input.int(10, "Pivot length (left = right)", minval=2, maxval=30, group=grpZ, tooltip="Zone is born only after this many bars confirm the pivot. Larger = stronger, fewer zones, later appearance. This delay is confirmation lag, NOT repainting.")
atrLen     = input.int(14, "ATR length", minval=1, group=grpZ)
halfW      = input.float(0.5, "Zone half-width (x ATR)", minval=0.1, maxval=1.5, step=0.05, group=grpZ, tooltip="Zone = level +/- half-width x ATR(14), frozen at the pivot bar.")
widthMode  = input.string("Centred", "Width mode", options=["Centred", "Inward"], group=grpZ, tooltip="Centred: level +/- half-width x ATR. Inward: resistance = [high - fuzz, high], support = [low, low + fuzz], fuzz = ATR(7)/2 x multiplier.")
maxZones   = input.int(8, "Max zones per side", minval=1, maxval=20, group=grpZ)
mergeDist  = input.float(0.5, "Merge distance (x ATR)", minval=0.0, step=0.1, group=grpZ, tooltip="A new pivot within this distance of an existing same-side zone reinforces it instead of drawing a duplicate.")
breakConfirm = input.int(1, "Break confirmation closes (1 or 2)", minval=1, maxval=2, group=grpZ, tooltip="1 = one body close past the far edge breaks the zone. 2 = needs a 2nd consecutive close.")
fuzzMult   = input.float(0.75, "Inward fuzz (x ATR(7)/2)", minval=0.1, step=0.05, group=grpZ)

useHtf1 = input.bool(true, "Show HTF 1 zones", group=grpH)
htf1    = input.timeframe("15", "HTF 1", group=grpH)
useHtf2 = input.bool(true, "Show HTF 2 zones", group=grpH)
htf2    = input.timeframe("60", "HTF 2", group=grpH)
useHtf3 = input.bool(true, "Show HTF 3 zones", group=grpH)
htf3    = input.timeframe("240", "HTF 3", group=grpH)

showLabels = input.bool(true, "Show zone labels", group=grpD)
showEvents = input.bool(true, "Show event labels (BOUNCE / BREAK / RETEST...)", group=grpD)
strongThr  = input.int(70, "Strong zone threshold (0-100)", minval=1, maxval=100, group=grpD)

alertTouch = input.bool(true, "Alert: touch of a STRONG zone", group=grpA)
alertBreak = input.bool(true, "Alert: body-close break of any zone", group=grpA)
alertRetest = input.bool(true, "Alert: retest held", group=grpA)

grpM  = "MTF-only view"

useChart = input.bool(true, "Chart timeframe counts (1 vote)", group=grpM, tooltip="ON: the chart timeframe itself counts as 1 vote toward agreement. OFF: a zone needs 2 checked HTF zones to reach the minimum 3.")
mtfOnly = input.bool(true, "MTF zones only (hide chart zones with no HTF agreement)", group=grpM, tooltip="ON: a chart-TF zone is drawn only where the CHECKED timeframes agree with it. OFF: all chart zones are drawn, as in v1.1. The engine still tracks hidden zones.")
minConf = input.int(3, "Minimum confluence to show (1 = chart, +2 per HTF)", minval=1, maxval=7, step=2, group=grpM, tooltip="3 = chart + 1 HTF (or 2 HTFs if the chart box is unchecked). 5 = chart + 2 HTFs. 7 = all 3 HTF zones agree.")

// ---------------- Colour settings ----------------
grpC  = "Colours"

resColIn  = input.color(color.red, "Resistance box colour", group=grpC)
supColIn  = input.color(color.green, "Support box colour", group=grpC)
flipColIn = input.color(color.orange, "Flipped zone colour", group=grpC)
deadColIn = input.color(color.gray, "Violated zone colour", group=grpC)
htfColIn  = input.color(color.blue, "HTF zone colour", group=grpC)
labTextIn = input.color(color.white, "Zone label text colour", group=grpC, tooltip="Text on the zone labels (and on the blue HTF labels).")

// ---------------- Colours (from the settings above) ----------------
resCol  = color.new(resColIn, 0)
resBg   = color.new(resColIn, 90)
supCol  = color.new(supColIn, 0)
supBg   = color.new(supColIn, 90)
flipCol = color.new(flipColIn, 0)
flipBg  = color.new(flipColIn, 88)
deadCol = color.new(deadColIn, 0)
deadBg  = color.new(deadColIn, 92)
htfCol  = color.new(htfColIn, 0)
htfBg   = color.new(htfColIn, 88)

// ---------------- State constants ----------------
ST_ACTIVE   = 0
ST_TOUCHED  = 1
ST_FLIPPED  = 2
ST_RETEST   = 3
ST_DONE_OK  = 4
ST_DONE_BAD = 5

// ---------------- Chart-TF zone storage (parallel arrays) ----------------
var array<float> zTop     = array.new_float()
var array<float> zBot     = array.new_float()
var array<float> zLvl     = array.new_float()
var array<int>   zSide    = array.new_int()
var array<int>   zState   = array.new_int()
var array<float> zTouches = array.new_float()
var array<int>   zHits    = array.new_int()
var array<int>   zPierces = array.new_int()
var array<int>   zBounces = array.new_int()
var array<int>   zBorn    = array.new_int()
var array<float> zVol     = array.new_float()
var array<box>   zBox     = array.new_box()
var array<label> zLab     = array.new_label()
var array<int>   zMark    = array.new_int()
var array<int>   zHold    = array.new_int()
var array<int>   zRetestBar = array.new_int()
var array<int>   zFirst   = array.new_int()
var array<bool>  zFlipped = array.new_bool()

// ---------------- HTF zone storage ----------------
var array<float> hTop  = array.new_float(6, na)
var array<float> hBot  = array.new_float(6, na)
var array<float> hLvl  = array.new_float(6, na)
var array<box>   hBox  = array.new_box(6, na)
var array<label> hLab  = array.new_label(6, na)
var array<float> hLastPh = array.new_float(3, na)
var array<float> hLastPl = array.new_float(3, na)

// ---------------- Series ----------------
atrVal  = ta.atr(atrLen)
atr7    = ta.atr(7)
volSMA  = ta.sma(volume, 50)
ph      = ta.pivothigh(high, pivLen, pivLen)
pl      = ta.pivotlow(low, pivLen, pivLen)

// Event flags (reset every bar) — feed alertcondition
var bool evTouchStrong = false
var bool evBreak = false
var bool evRetestHeld = false
evTouchStrong := false
evBreak := false
evRetestHeld := false

// ---------------- Helpers ----------------
zoneCount(_side) =>
    int n = 0
    int sz = array.size(zSide)
    if sz > 0
        for i = 0 to sz - 1
            if array.get(zSide, i) == _side
                n += 1
    n

strengthOf(_i) =>
    float volS = 0.5
    float vAt = array.get(zVol, _i)
    if not na(volSMA) and volSMA > 0 and vAt > 0
        volS := math.min(vAt / volSMA, 3.0) / 3.0
    float tS = math.min(array.get(zTouches, _i) / 5.0, 1.0)
    float aS = math.max(1.0 - (bar_index - array.get(zBorn, _i)) / 500.0, 0.0)
    0.40 * volS + 0.30 * tS + 0.30 * aS

confluenceOf(_i) =>
    float top = array.get(zTop, _i)
    float bot = array.get(zBot, _i)
    int c = useChart ? 1 : 0
    for s = 0 to 5
        float ht = array.get(hTop, s)
        float hb = array.get(hBot, s)
        if not na(ht) and not na(hb) and ht >= bot and hb <= top
            c += 2
    c

// A chart zone is VISIBLE when MTF-only is off, or its confluence reaches the minimum.
zoneVisible(_i) =>
    not mtfOnly or confluenceOf(_i) >= minConf

labelText(_i) =>
    string side = array.get(zSide, _i) == 1 ? "R" : "S"
    string lvl = str.tostring(array.get(zLvl, _i), format.mintick)
    int sc = math.round(strengthOf(_i) * 100)
    int conf = confluenceOf(_i)
    string weak = array.get(zPierces, _i) >= 3 ? " WEAK" : ""
    string flip = array.get(zFlipped, _i) ? " FLIP" : ""
    string htfTxt = conf > 1 ? " HTF" : ""
    side + " " + lvl + " · T" + str.tostring(array.get(zHits, _i)) + " · P" + str.tostring(array.get(zPierces, _i)) + " · " + str.tostring(sc) + "% · x" + str.tostring(conf) + weak + flip + htfTxt

refreshLabel(_i) =>
    box b = array.get(zBox, _i)
    label l = array.get(zLab, _i)
    int sc = math.round(strengthOf(_i) * 100)
    bool vis = zoneVisible(_i)
    color bg = array.get(zSide, _i) == 1 ? resBg : supBg
    color bcol = array.get(zSide, _i) == 1 ? resCol : supCol
    if array.get(zState, _i) == ST_DONE_BAD
        bg := deadBg
        bcol := deadCol
    else if array.get(zFlipped, _i)
        bg := flipBg
        bcol := flipCol
    if not vis
        // hidden zone: fully transparent box + label (engine still tracks it)
        box.set_border_color(b, color.new(color.white, 100))
        box.set_bgcolor(b, color.new(color.white, 100))
        label.set_color(l, color.new(color.white, 100))
        label.set_textcolor(l, color.new(color.white, 100))
    else
        box.set_border_color(b, bcol)
        box.set_bgcolor(b, bg)
        label.set_color(l, bg)
        label.set_textcolor(l, sc >= strongThr ? labTextIn : color.new(labTextIn, 20))
    if showLabels
        label.set_text(l, labelText(_i))
        label.set_xy(l, bar_index, array.get(zSide, _i) == 1 ? array.get(zTop, _i) : array.get(zBot, _i))

removeZone(_i) =>
    box b = array.get(zBox, _i)
    label l = array.get(zLab, _i)
    if not na(b)
        box.delete(b)
    if not na(l)
        label.delete(l)
    array.remove(zTop, _i)
    array.remove(zBot, _i)
    array.remove(zLvl, _i)
    array.remove(zSide, _i)
    array.remove(zState, _i)
    array.remove(zTouches, _i)
    array.remove(zHits, _i)
    array.remove(zPierces, _i)
    array.remove(zBounces, _i)
    array.remove(zBorn, _i)
    array.remove(zVol, _i)
    array.remove(zBox, _i)
    array.remove(zLab, _i)
    array.remove(zMark, _i)
    array.remove(zHold, _i)
    array.remove(zRetestBar, _i)
    array.remove(zFirst, _i)
    array.remove(zFlipped, _i)

addZone(_side, _level, _vol) =>
    // width — ATR frozen at the PIVOT bar so the zone never breathes
    float top = na
    float bot = na
    if widthMode == "Centred"
        top := _level + halfW * atrVal[pivLen]
        bot := _level - halfW * atrVal[pivLen]
    else
        float fuzz = atr7[pivLen] / 2 * fuzzMult
        if _side == 1
            top := _level
            bot := _level - fuzz
        else
            top := _level + fuzz
            bot := _level
    // merge: reinforce an existing same-side zone within mergeDist x ATR
    bool merged = false
    int nz = array.size(zLvl)
    if nz > 0
        for i = 0 to nz - 1
            if not merged and array.get(zSide, i) == _side and math.abs(array.get(zLvl, i) - _level) <= mergeDist * atrVal
                array.set(zTouches, i, array.get(zTouches, i) + 1.0)
                array.set(zHits, i, array.get(zHits, i) + 1)
                merged := true
                refreshLabel(i)
    if not merged
        // too-close filter: same-side zones within 1 x ATR -> keep the nearer to price
        int dropClose = -1
        bool suppressNew = false
        nz := array.size(zLvl)
        if nz > 0
            for i = 0 to nz - 1
                if array.get(zSide, i) == _side and math.abs(array.get(zLvl, i) - _level) <= atrVal
                    if math.abs(array.get(zLvl, i) - close) < math.abs(_level - close)
                        suppressNew := true
                    else if dropClose == -1
                        dropClose := i
        if not suppressNew
            if dropClose >= 0
                removeZone(dropClose)
            // capacity: drop the oldest UNTESTED same-side zone first
            if zoneCount(_side) >= maxZones
                int drop = -1
                int oldest = 1000000000
                int sz2 = array.size(zSide)
                if sz2 > 0
                    for i = 0 to sz2 - 1
                        if array.get(zSide, i) == _side and array.get(zHits, i) == 0 and array.get(zBorn, i) < oldest
                            oldest := array.get(zBorn, i)
                            drop := i
                    if drop == -1
                        int oldestAny = 1000000000
                        for i = 0 to sz2 - 1
                            if array.get(zSide, i) == _side and array.get(zBorn, i) < oldestAny
                                oldestAny := array.get(zBorn, i)
                                drop := i
                if drop >= 0
                    removeZone(drop)
            color bcol = _side == 1 ? resCol : supCol
            color bbg = _side == 1 ? resBg : supBg
            box nb = box.new(bar_index, top, bar_index + 1, bot, xloc=xloc.bar_index, extend=extend.right, border_color=bcol, bgcolor=bbg)
            label nl = label.new(bar_index, _side == 1 ? top : bot, "", xloc=xloc.bar_index, style=_side == 1 ? label.style_label_down : label.style_label_up, color=bbg, textcolor=labTextIn, size=size.small)
            array.push(zTop, top)
            array.push(zBot, bot)
            array.push(zLvl, _level)
            array.push(zSide, _side)
            array.push(zState, ST_ACTIVE)
            array.push(zTouches, 0.0)
            array.push(zHits, 0)
            array.push(zPierces, 0)
            array.push(zBounces, 0)
            array.push(zBorn, bar_index)
            array.push(zVol, _vol)
            array.push(zBox, nb)
            array.push(zLab, nl)
            array.push(zMark, int(na))
            array.push(zHold, 0)
            array.push(zRetestBar, int(na))
            array.push(zFirst, 0)
            array.push(zFlipped, false)
            refreshLabel(array.size(zLvl) - 1)

// ---------------- HTF zone update (latest zone per side per slot) ----------------
htfUpdate(_slot, _ph, _pl, _atr, _tf) =>
    // resistance
    if not na(_ph) and not na(_atr) and (na(array.get(hLastPh, _slot)) or array.get(hLastPh, _slot) != _ph)
        array.set(hLastPh, _slot, _ph)
        int idxR = _slot * 2
        float topR = _ph + halfW * _atr
        float botR = _ph - halfW * _atr
        box obR = array.get(hBox, idxR)
        label olR = array.get(hLab, idxR)
        if not na(obR)
            box.delete(obR)
        if not na(olR)
            label.delete(olR)
        box nbR = box.new(bar_index, topR, bar_index + 1, botR, xloc=xloc.bar_index, extend=extend.right, border_color=htfCol, bgcolor=htfBg)
        label nlR = label.new(bar_index, topR, "HTF " + _tf + " R " + str.tostring(_ph, format.mintick), xloc=xloc.bar_index, style=label.style_label_down, color=htfBg, textcolor=labTextIn, size=size.small)
        array.set(hTop, idxR, topR)
        array.set(hBot, idxR, botR)
        array.set(hLvl, idxR, _ph)
        array.set(hBox, idxR, nbR)
        array.set(hLab, idxR, nlR)
    // support
    if not na(_pl) and not na(_atr) and (na(array.get(hLastPl, _slot)) or array.get(hLastPl, _slot) != _pl)
        array.set(hLastPl, _slot, _pl)
        int idxS = _slot * 2 + 1
        float topS = _pl + halfW * _atr
        float botS = _pl - halfW * _atr
        box obS = array.get(hBox, idxS)
        label olS = array.get(hLab, idxS)
        if not na(obS)
            box.delete(obS)
        if not na(olS)
            label.delete(olS)
        box nbS = box.new(bar_index, topS, bar_index + 1, botS, xloc=xloc.bar_index, extend=extend.right, border_color=htfCol, bgcolor=htfBg)
        label nlS = label.new(bar_index, botS, "HTF " + _tf + " S " + str.tostring(_pl, format.mintick), xloc=xloc.bar_index, style=label.style_label_up, color=htfBg, textcolor=labTextIn, size=size.small)
        array.set(hTop, idxS, topS)
        array.set(hBot, idxS, botS)
        array.set(hLvl, idxS, _pl)
        array.set(hBox, idxS, nbS)
        array.set(hLab, idxS, nlS)

// ---------------- HTF data (maths INSIDE the HTF context, lookahead OFF) ----------------
[hp1, lp1, at1] = request.security(syminfo.tickerid, htf1, [ta.pivothigh(high, pivLen, pivLen), ta.pivotlow(low, pivLen, pivLen), ta.atr(atrLen)[pivLen]], lookahead=barmerge.lookahead_off)
[hp2, lp2, at2] = request.security(syminfo.tickerid, htf2, [ta.pivothigh(high, pivLen, pivLen), ta.pivotlow(low, pivLen, pivLen), ta.atr(atrLen)[pivLen]], lookahead=barmerge.lookahead_off)
[hp3, lp3, at3] = request.security(syminfo.tickerid, htf3, [ta.pivothigh(high, pivLen, pivLen), ta.pivotlow(low, pivLen, pivLen), ta.atr(atrLen)[pivLen]], lookahead=barmerge.lookahead_off)

chartSec = timeframe.in_seconds()
ok1 = useHtf1 and timeframe.in_seconds(htf1) > chartSec
ok2 = useHtf2 and timeframe.in_seconds(htf2) > chartSec
ok3 = useHtf3 and timeframe.in_seconds(htf3) > chartSec

// ---------------- Main logic — CLOSED BARS ONLY (no repaint) ----------------
if barstate.isconfirmed and not na(atrVal)
    // HTF zones first (so confluence counts them)
    if ok1
        htfUpdate(0, hp1, lp1, at1, htf1)
    if ok2
        htfUpdate(1, hp2, lp2, at2, htf2)
    if ok3
        htfUpdate(2, hp3, lp3, at3, htf3)

    // HTF zones moved: re-evaluate which chart zones the MTF-only view shows
    if mtfOnly
        int nz0 = array.size(zLvl)
        if nz0 > 0
            for i = 0 to nz0 - 1
                refreshLabel(i)

    // New chart-TF zones on confirmed pivots (volume at the PIVOT bar)
    if not na(ph)
        addZone(1, ph, nz(volume[pivLen], 0))
    if not na(pl)
        addZone(-1, pl, nz(volume[pivLen], 0))

    // Zone state machines
    int nZones = array.size(zLvl)
    if nZones > 0
        for i = 0 to nZones - 1
            int side = array.get(zSide, i)
            int st = array.get(zState, i)
            float top = array.get(zTop, i)
            float bot = array.get(zBot, i)
            bool overlaps = low <= top and high >= bot
            bool hit = close >= bot and close <= top and not (close[1] >= bot and close[1] <= top)
            bool pierce = overlaps and not (close >= bot and close <= top)
            bool bodyBreak = side == 1 ? close > top : close < bot
            bool bodyBreakPrev = side == 1 ? close[1] > top : close[1] < bot
            bool bodyBreakConf = bodyBreak and (breakConfirm == 1 or bodyBreakPrev)
            bool closeOutsideNear = side == 1 ? close < bot : close > top
            bool vis = zoneVisible(i)

            if st == ST_ACTIVE
                if hit
                    // touch quality: false break 1.2 / body 0.8 / consolidation 0.5
                    float q = 0.8
                    if side == 1 and high > top and close <= top
                        q := 1.2
                    else if side == -1 and low < bot and close >= bot
                        q := 1.2
                    else if open >= bot and open <= top and close >= bot and close <= top
                        q := 0.5
                    array.set(zTouches, i, array.get(zTouches, i) + q)
                    array.set(zHits, i, array.get(zHits, i) + 1)
                    array.set(zState, i, ST_TOUCHED)
                    array.set(zMark, i, bar_index)
                    if strengthOf(i) * 100 >= strongThr and vis
                        evTouchStrong := true
                    refreshLabel(i)
                else if pierce
                    array.set(zPierces, i, array.get(zPierces, i) + 1)
                    refreshLabel(i)
                // break can also happen without a prior hit on this zone
                if bodyBreakConf
                    if showEvents and vis
                        label.new(bar_index, side == 1 ? high : low, "BREAK", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 0), textcolor=color.white, size=size.small)
                    if vis
                        evBreak := true
                    array.set(zState, i, ST_FLIPPED)
                    array.set(zMark, i, bar_index)
                    array.set(zSide, i, -side)
                    array.set(zFlipped, i, true)
                    array.set(zHits, i, 0)
                    if array.get(zFirst, i) == 0
                        array.set(zFirst, i, 2)
                    box b1 = array.get(zBox, i)
                    box.set_border_color(b1, flipCol)
                    box.set_bgcolor(b1, flipBg)
                    refreshLabel(i)

            else if st == ST_TOUCHED
                int markBar = nz(array.get(zMark, i), bar_index)
                if closeOutsideNear and bar_index - markBar <= 3
                    // BOUNCE: close back outside within 3 bars
                    if showEvents and vis
                        label.new(bar_index, side == 1 ? top : bot, "BOUNCE", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.green, 0), textcolor=color.white, size=size.small)
                    array.set(zBounces, i, array.get(zBounces, i) + 1)
                    if array.get(zFirst, i) == 0
                        array.set(zFirst, i, 1)
                    array.set(zState, i, ST_ACTIVE)
                    array.set(zMark, i, int(na))
                    refreshLabel(i)
                else if bodyBreakConf
                    if showEvents and vis
                        label.new(bar_index, side == 1 ? high : low, "BREAK", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 0), textcolor=color.white, size=size.small)
                    if vis
                        evBreak := true
                    array.set(zState, i, ST_FLIPPED)
                    array.set(zMark, i, bar_index)
                    array.set(zSide, i, -side)
                    array.set(zFlipped, i, true)
                    array.set(zHits, i, 0)
                    if array.get(zFirst, i) == 0
                        array.set(zFirst, i, 2)
                    box b2 = array.get(zBox, i)
                    box.set_border_color(b2, flipCol)
                    box.set_bgcolor(b2, flipBg)
                    refreshLabel(i)
                else if bar_index - markBar > 3
                    // touch went stale: zone stays active
                    array.set(zState, i, ST_ACTIVE)
                    array.set(zMark, i, int(na))

            else if st == ST_FLIPPED
                int markBar2 = nz(array.get(zMark, i), bar_index)
                if bar_index - markBar2 > 20
                    // break with no retest in 20 bars = violated
                    if showEvents and vis
                        label.new(bar_index, side == 1 ? top : bot, "NO RETEST", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.gray, 0), textcolor=color.white, size=size.small)
                    array.set(zState, i, ST_DONE_BAD)
                    box b3 = array.get(zBox, i)
                    box.set_border_color(b3, deadCol)
                    box.set_bgcolor(b3, deadBg)
                    refreshLabel(i)
                else if overlaps and bar_index > markBar2
                    // retest: price returned into the flipped zone
                    if showEvents and vis
                        label.new(bar_index, side == 1 ? bot : top, "RETEST", style=side == 1 ? label.style_label_up : label.style_label_down, color=color.new(color.orange, 0), textcolor=color.white, size=size.small)
                    array.set(zState, i, ST_RETEST)
                    array.set(zRetestBar, i, bar_index)
                    array.set(zHold, i, 0)

            else if st == ST_RETEST
                int rtBar = nz(array.get(zRetestBar, i), bar_index)
                if bar_index > rtBar
                    bool failed = side == 1 ? close < bot : close > top
                    if failed
                        if showEvents and vis
                            label.new(bar_index, side == 1 ? high : low, "RETEST FAILED", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 0), textcolor=color.white, size=size.small)
                        array.set(zState, i, ST_DONE_BAD)
                        box b4 = array.get(zBox, i)
                        box.set_border_color(b4, deadCol)
                        box.set_bgcolor(b4, deadBg)
                        refreshLabel(i)
                    else
                        array.set(zHold, i, array.get(zHold, i) + 1)
                        if array.get(zHold, i) >= 3
                            if showEvents and vis
                                label.new(bar_index, side == 1 ? bot : top, "HOLD ✓", style=side == 1 ? label.style_label_up : label.style_label_down, color=color.new(color.green, 0), textcolor=color.white, size=size.small)
                            if vis
                                evRetestHeld := true
                            array.set(zState, i, ST_ACTIVE)
                            array.set(zMark, i, int(na))
                            refreshLabel(i)

    // keep labels glued to the right edge
    int nLab = array.size(zLab)
    if nLab > 0
        for i = 0 to nLab - 1
            label.set_xy(array.get(zLab, i), bar_index, array.get(zSide, i) == 1 ? array.get(zTop, i) : array.get(zBot, i))
    for s = 0 to 5
        label ol = array.get(hLab, s)
        if not na(ol)
            label.set_xy(ol, bar_index, s % 2 == 0 ? array.get(hTop, s) : array.get(hBot, s))
