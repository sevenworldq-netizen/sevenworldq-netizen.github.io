//@version=5
// ============================================================================
// AMANDIRE S&R ZONE DETECTOR — v1.0 (28 Nov 2026)
// A first-of-its-kind S&R zone detector built from Trader Noah's own measured
// research (Mercury study, MTF Zones v2.0/v2.1, 65-page all-videos-and-bots
// report) + published non-repainting methods.
//
// WHAT IT DOES
//   * Draws support / resistance zones from CONFIRMED pivots only.
//   * Counts touch EVENTS (close enters the zone) and wick PIERCES separately.
//   * Scores every zone 0-100 (volume 40% / touches 30% / age 30%).
//   * Labels touches, bounces, body-close breaks, flip (turncoat) zones,
//     retests and holds. Flags weakening zones (>= 3 pierces).
//   * Overlays higher-timeframe zones (15m / 1H / 4H) and scores confluence.
//   * Alerts on: touch of a STRONG zone, body-close break, retest-held.
//
// HONEST EXPECTATIONS — READ THIS
//   * NO indicator detects zones 100%. Published measurements + our own Mercury
//     study put zone respect at roughly 40-75%, and random bands hold about as
//     often as real zones. Respect is NOT a trading edge: Mercury's best
//     indicator (72.57% respected) LOSES money on 1m on both BTC and gold.
//   * A zone appears only after its pivot CONFIRMS (pivot-length bars late).
//     That delay is confirmation lag, NOT repainting. Zero lag is impossible:
//     any "zone" known earlier is a guess that can be redrawn.
//   * A touch alone is NEVER an entry (Selam's rule: sweep + rejection + MSS
//     required). This indicator marks zones; the entry rules sit on top.
//
// NON-REPAINTING CONTRACT
//   1. Zones are born ONLY from ta.pivothigh / ta.pivotlow (confirmed pivots).
//   2. Once drawn, a zone NEVER moves. Level/top/bottom are frozen at the
//      confirmation bar (ATR frozen at the pivot bar).
//   3. All event detection runs on CLOSED bars only (barstate.isconfirmed),
//      so a live-bar label can never appear and vanish.
//   4. HTF data via request.security(..., lookahead_off), maths computed
//      INSIDE the HTF context. Never lookahead_on without an offset.
//   5. A break counts ONLY on a candle BODY close past the zone's far edge.
//      A wick through is a sweep/pierce — the zone stays alive.
//   6. A broken zone is never silently erased: it flips role (turncoat),
//      turns amber, and is watched for a retest. History is never rewritten.
//   7. No barstate.islast price labelling or signals anywhere (that
//      construction repaints). The info table is display-only current
//      counts, not a signal.
//
// HOW TO USE
//   Paste into the TradingView Pine Editor, "Add to chart". Default settings
//   are the measured defaults (pivot 10, half-width 0.5 x ATR(14)).
//   Strong zones (>= 70) are the most opaque. "HTF" badge = overlaps a
//   higher-timeframe zone. "CONF xN" = confluence count (chart TF = 1,
//   each HTF = 2). Weakening = 3+ wick pierces: trade the BREAK, not the bounce.
// ============================================================================

indicator("Amandire S&R Zone Detector", shorttitle="Amandire SR Zones", overlay=true, max_boxes_count=500, max_labels_count=500, max_lines_count=500)

// ---------------- Inputs ----------------
grpZ  = "Zones"
grpH  = "Higher timeframes (MTF)"
grpD  = "Display"
grpA  = "Alerts"

pivLen     = input.int(10, "Pivot length (left = right)", minval=2, maxval=30, group=grpZ, tooltip="Zone is born only after this many bars confirm the pivot. Larger = stronger, fewer zones, later appearance. This delay is confirmation lag, NOT repainting.")
atrLen     = input.int(14, "ATR length", minval=1, group=grpZ)
halfW      = input.float(0.5, "Zone half-width (x ATR)", minval=0.1, maxval=1.5, step=0.05, group=grpZ, tooltip="Zone = level +/- half-width x ATR(14), frozen at the pivot bar. Total width = 2 x this. Wider bands respect more often by construction — report the width with any respect rate.")
widthMode  = input.string("Centred", "Width mode", options=["Centred", "Inward"], group=grpZ, tooltip="Centred: level +/- half-width x ATR. Inward (Selam-engine style): resistance = [high - fuzz, high], support = [low, low + fuzz], fuzz = ATR(7)/2 x 0.75.")
maxZones   = input.int(8, "Max zones per side", minval=1, maxval=20, group=grpZ)
mergeDist  = input.float(0.5, "Merge distance (x ATR)", minval=0.0, step=0.1, group=grpZ, tooltip="A new pivot within this distance of an existing same-side zone reinforces it (touch +1) instead of drawing a duplicate.")
breakConfirm = input.int(1, "Break confirmation closes (1 or 2)", minval=1, maxval=2, group=grpZ, tooltip="1 = Selam's rule: one body close past the far edge breaks the zone. 2 = needs a 2nd consecutive close. Measured on gold 1m: false breaks fall from ~51% to ~41%, but ~26% of breaks never confirm.")
fuzzMult   = input.float(0.75, "Inward fuzz (x ATR(7)/2)", minval=0.1, step=0.05, group=grpZ)

useHtf1 = input.bool(true, "Show HTF 1 zones", group=grpH)
htf1    = input.timeframe("15", "HTF 1", group=grpH)
useHtf2 = input.bool(true, "Show HTF 2 zones", group=grpH)
htf2    = input.timeframe("60", "HTF 2", group=grpH)
useHtf3 = input.bool(true, "Show HTF 3 zones", group=grpH)
htf3    = input.timeframe("240", "HTF 3", group=grpH)

showLabels = input.bool(true, "Show zone labels", group=grpD)
showEvents = input.bool(true, "Show event labels (BOUNCE / BREAK / RETEST...)", group=grpD)
strongThr  = input.int(70, "Strong zone threshold (0-100)", minval=1, maxval=100, group=grpD)

alertTouch = input.bool(true, "Alert: touch of a STRONG zone", group=grpA)
alertBreak = input.bool(true, "Alert: body-close break of any zone", group=grpA)
alertRetest = input.bool(true, "Alert: retest held", group=grpA)

// ---------------- Colours ----------------
resCol  = color.new(color.red, 0)
resBg   = color.new(color.red, 90)
supCol  = color.new(color.green, 0)
supBg   = color.new(color.green, 90)
flipCol = color.new(color.orange, 0)
flipBg  = color.new(color.orange, 88)
deadCol = color.new(color.gray, 0)
deadBg  = color.new(color.gray, 92)
htfCol  = color.new(color.blue, 0)

// ---------------- State constants ----------------
ST_ACTIVE = 0   // live zone, waiting
ST_TOUCHED = 1  // hit within the last 3 bars, bounce/break pending
ST_FLIPPED = 2  // broken, role reversed, retest watch open (20 bars)
ST_RETEST = 3   // retest in progress, hold watch (3 bars)
ST_DONE_OK = 4  // respected (bounce or retest held) — zone stays live
ST_DONE_BAD = 5 // violated (retest failed / no retest) — greyed, kept visible

// ---------------- Chart-TF zone storage (parallel arrays) ----------------
var array<float> zTop     = array.new_float()
var array<float> zBot     = array.new_float()
var array<float> zLvl     = array.new_float()
var array<int>   zSide    = array.new_int()    // 1 resistance, -1 support
var array<int>   zState   = array.new_int()
var array<float> zTouches = array.new_float()  // quality-weighted touches
var array<int>   zHits    = array.new_int()    // touch events (close entered)
var array<int>   zPierces = array.new_int()    // wick-only entries
var array<int>   zBounces = array.new_int()
var array<int>   zBorn    = array.new_int()    // bar_index of confirmation
var array<float> zVol     = array.new_float()  // volume at pivot bar
var array<box>   zBox     = array.new_box()
var array<label> zLab     = array.new_label()
var array<int>   zMark    = array.new_int()    // bar of last touch / break
var array<int>   zHold    = array.new_int()    // hold-bar counter
var array<int>   zRetestBar = array.new_int()
var array<int>   zFirst   = array.new_int()    // first outcome: 0 none, 1 bounce, 2 break
var array<bool>  zFlipped = array.new_bool()   // role already reversed once

// ---------------- HTF zone storage: latest zone per side per slot (3 slots x 2 sides) ----------------
var array<float> hTop  = array.new_float(6, na)
var array<float> hBot  = array.new_float(6, na)
var array<float> hLvl  = array.new_float(6, na)
var array<box>   hBox  = array.new_box(6, na)
var array<label> hLab  = array.new_label(6, na)
var array<float> hLastPh = array.new_float(3, na)
var array<float> hLastPl = array.new_float(3, na)

// ---------------- Series ----------------
atrVal  = ta.atr(atrLen)
atr7    = ta.atr(7)
volSMA  = ta.sma(volume, 50)
ph      = ta.pivothigh(high, pivLen, pivLen)
pl      = ta.pivotlow(low, pivLen, pivLen)

// Event flags (reset every bar, set only on confirmed bars) — feed alertcondition
var bool evTouchStrong = false
var bool evBreak = false
var bool evRetestHeld = false
evTouchStrong := false
evBreak := false
evRetestHeld := false

// ---------------- Helpers ----------------
zoneCount(_side) =>
    int n = 0
    for i = 0 to array.size(zSide) - 1
        if array.get(zSide, i) == _side
            n += 1
    n

strengthOf(_i) =>
    float volS = 0.5
    float vAt = array.get(zVol, _i)
    if not na(volSMA) and volSMA > 0 and vAt > 0
        volS := math.min(vAt / volSMA, 3.0) / 3.0
    float tS = math.min(array.get(zTouches, _i) / 5.0, 1.0)
    float aS = math.max(1.0 - (bar_index - array.get(zBorn, _i)) / 500.0, 0.0)
    0.40 * volS + 0.30 * tS + 0.30 * aS

confluenceOf(_i) =>
    float top = array.get(zTop, _i)
    float bot = array.get(zBot, _i)
    int c = 1
    for s = 0 to 5
        float ht = array.get(hTop, s)
        float hb = array.get(hBot, s)
        if not na(ht) and not na(hb) and ht >= bot and hb <= top
            c += 2
    c

labelText(_i) =>
    string side = array.get(zSide, _i) == 1 ? "R" : "S"
    string lvl = str.tostring(array.get(zLvl, _i), format.mintick)
    int str = math.round(strengthOf(_i) * 100)
    int conf = confluenceOf(_i)
    string weak = array.get(zPierces, _i) >= 3 ? " WEAK" : ""
    string flip = array.get(zFlipped, _i) ? " FLIP" : ""
    string htf = conf > 1 ? " HTF" : ""
    side + " " + lvl + " · T" + str.tostring(array.get(zHits, _i)) + " · P" + str.tostring(array.get(zPierces, _i)) + " · " + str.tostring(str) + "% · x" + str.tostring(conf) + weak + flip + htf

refreshLabel(_i) =>
    if showLabels
        label l = array.get(zLab, _i)
        int str = math.round(strengthOf(_i) * 100)
        color bg = array.get(zSide, _i) == 1 ? resBg : supBg
        if array.get(zState, _i) == ST_DONE_BAD
            bg := deadBg
        else if array.get(zFlipped, _i)
            bg := flipBg
        label.set_text(l, labelText(_i))
        label.set_xy(l, bar_index, array.get(zSide, _i) == 1 ? array.get(zTop, _i) : array.get(zBot, _i))
        label.set_color(l, bg)
        label.set_textcolor(l, str >= strongThr ? color.white : color.new(color.white, 20))

removeZone(_i) =>
    box b = array.get(zBox, _i)
    label l = array.get(zLab, _i)
    if not na(b)
        box.delete(b)
    if not na(l)
        label.delete(l)
    array.remove(zTop, _i)
    array.remove(zBot, _i)
    array.remove(zLvl, _i)
    array.remove(zSide, _i)
    array.remove(zState, _i)
    array.remove(zTouches, _i)
    array.remove(zHits, _i)
    array.remove(zPierces, _i)
    array.remove(zBounces, _i)
    array.remove(zBorn, _i)
    array.remove(zVol, _i)
    array.remove(zBox, _i)
    array.remove(zLab, _i)
    array.remove(zMark, _i)
    array.remove(zHold, _i)
    array.remove(zRetestBar, _i)
    array.remove(zFirst, _i)
    array.remove(zFlipped, _i)

addZone(_side, _level, _vol) =>
    // width — ATR frozen at the PIVOT bar (atrVal[pivLen]), not the
    // confirmation bar, so the zone never breathes after it is drawn
    float top = na
    float bot = na
    if widthMode == "Centred"
        top := _level + halfW * atrVal[pivLen]
        bot := _level - halfW * atrVal[pivLen]
    else
        float fuzz = atr7[pivLen] / 2 * fuzzMult
        if _side == 1
            top := _level
            bot := _level - fuzz
        else
            top := _level + fuzz
            bot := _level
    // merge: reinforce an existing same-side zone within mergeDist x ATR
    bool merged = false
    for i = 0 to array.size(zLvl) - 1
        if not merged and array.get(zSide, i) == _side and math.abs(array.get(zLvl, i) - _level) <= mergeDist * atrVal
            array.set(zTouches, i, array.get(zTouches, i) + 1.0)
            array.set(zHits, i, array.get(zHits, i) + 1)
            merged := true
            refreshLabel(i)
    if not merged
        // too-close filter FIRST (WFRX / Zone Bell rule): same-side zones
        // within 1 x ATR(14) -> keep the nearer to price. If the new zone
        // is the farther one, suppress it entirely (nothing is removed).
        int dropClose = -1
        for i = 0 to array.size(zLvl) - 1
            if array.get(zSide, i) == _side and math.abs(array.get(zLvl, i) - _level) <= atrVal
                if math.abs(array.get(zLvl, i) - close) < math.abs(_level - close)
                    dropClose := -2  // the new one is farther: do not add it
                else
                    dropClose := i   // the old one is farther: drop it
        if dropClose != -2
            if dropClose >= 0
                removeZone(dropClose)
            // capacity: drop the oldest UNTESTED same-side zone first
            if zoneCount(_side) >= maxZones
                int drop = -1
                int oldest = 1000000000
                for i = 0 to array.size(zSide) - 1
                    if array.get(zSide, i) == _side and array.get(zHits, i) == 0 and array.get(zBorn, i) < oldest
                        oldest := array.get(zBorn, i)
                        drop := i
                if drop == -1
                    int oldestAny = 1000000000
                    for i = 0 to array.size(zSide) - 1
                        if array.get(zSide, i) == _side and array.get(zBorn, i) < oldestAny
                            oldestAny := array.get(zBorn, i)
                            drop := i
                if drop >= 0
                    removeZone(drop)
            color bcol = _side == 1 ? resCol : supCol
            color bbg = _side == 1 ? resBg : supBg
            box nb = box.new(bar_index, top, bar_index + 1, bot, xloc=xloc.bar_index, extend=extend.right, border_color=bcol, bgcolor=bbg)
            label nl = label.new(bar_index, _side == 1 ? top : bot, "", xloc=xloc.bar_index, style=_side == 1 ? label.style_label_down : label.style_label_up, color=bbg, textcolor=color.white, size=size.small)
            array.push(zTop, top)
            array.push(zBot, bot)
            array.push(zLvl, _level)
            array.push(zSide, _side)
            array.push(zState, ST_ACTIVE)
            array.push(zTouches, 0.0)
            array.push(zHits, 0)
            array.push(zPierces, 0)
            array.push(zBounces, 0)
            array.push(zBorn, bar_index)
            array.push(zVol, _vol)
            array.push(zBox, nb)
            array.push(zLab, nl)
            array.push(zMark, na)
            array.push(zHold, 0)
            array.push(zRetestBar, na)
            array.push(zFirst, 0)
            array.push(zFlipped, false)
            refreshLabel(array.size(zLvl) - 1)

// ---------------- HTF zone update (latest zone per side per slot) ----------------
htfUpdate(_slot, _ph, _pl, _atr, _tf) =>
    // resistance
    if not na(_ph) and not na(_atr) and (na(array.get(hLastPh, _slot)) or array.get(hLastPh, _slot) != _ph)
        array.set(hLastPh, _slot, _ph)
        int idx = _slot * 2
        float top = _ph + halfW * _atr
        float bot = _ph - halfW * _atr
        box ob = array.get(hBox, idx)
        label ol = array.get(hLab, idx)
        if not na(ob)
            box.delete(ob)
        if not na(ol)
            label.delete(ol)
        box nb = box.new(bar_index, top, bar_index + 1, bot, xloc=xloc.bar_index, extend=extend.right, border_color=htfCol, bgcolor=color.new(color.blue, 88))
        label nl = label.new(bar_index, top, "HTF " + _tf + " R " + str.tostring(_ph, format.mintick), xloc=xloc.bar_index, style=label.style_label_down, color=color.new(color.blue, 88), textcolor=color.white, size=size.small)
        array.set(hTop, idx, top)
        array.set(hBot, idx, bot)
        array.set(hLvl, idx, _ph)
        array.set(hBox, idx, nb)
        array.set(hLab, idx, nl)
    // support
    if not na(_pl) and not na(_atr) and (na(array.get(hLastPl, _slot)) or array.get(hLastPl, _slot) != _pl)
        array.set(hLastPl, _slot, _pl)
        int idx = _slot * 2 + 1
        float top = _pl + halfW * _atr
        float bot = _pl - halfW * _atr
        box ob = array.get(hBox, idx)
        label ol = array.get(hLab, idx)
        if not na(ob)
            box.delete(ob)
        if not na(ol)
            label.delete(ol)
        box nb = box.new(bar_index, top, bar_index + 1, bot, xloc=xloc.bar_index, extend=extend.right, border_color=htfCol, bgcolor=color.new(color.blue, 88))
        label nl = label.new(bar_index, bot, "HTF " + _tf + " S " + str.tostring(_pl, format.mintick), xloc=xloc.bar_index, style=label.style_label_up, color=color.new(color.blue, 88), textcolor=color.white, size=size.small)
        array.set(hTop, idx, top)
        array.set(hBot, idx, bot)
        array.set(hLvl, idx, _pl)
        array.set(hBox, idx, nb)
        array.set(hLab, idx, nl)

// ---------------- HTF data (pivot maths INSIDE the HTF context, lookahead OFF;
// ATR indexed back to the HTF pivot bar, matching the chart-TF rule) ----------------
[hp1, lp1, at1] = request.security(syminfo.tickerid, htf1, [ta.pivothigh(high, pivLen, pivLen), ta.pivotlow(low, pivLen, pivLen), ta.atr(atrLen)[pivLen]], lookahead=barmerge.lookahead_off)
[hp2, lp2, at2] = request.security(syminfo.tickerid, htf2, [ta.pivothigh(high, pivLen, pivLen), ta.pivotlow(low, pivLen, pivLen), ta.atr(atrLen)[pivLen]], lookahead=barmerge.lookahead_off)
[hp3, lp3, at3] = request.security(syminfo.tickerid, htf3, [ta.pivothigh(high, pivLen, pivLen), ta.pivotlow(low, pivLen, pivLen), ta.atr(atrLen)[pivLen]], lookahead=barmerge.lookahead_off)

// ---------------- Main logic — CLOSED BARS ONLY (no repaint) ----------------
if barstate.isconfirmed and not na(atrVal)
    // HTF zones first (so confluence counts them)
    if useHtf1
        htfUpdate(0, hp1, lp1, at1, htf1)
    if useHtf2
        htfUpdate(1, hp2, lp2, at2, htf2)
    if useHtf3
        htfUpdate(2, hp3, lp3, at3, htf3)

    // New chart-TF zones on confirmed pivots (volume at the PIVOT bar)
    if not na(ph)
        addZone(1, ph, nz(volume[pivLen], 0))
    if not na(pl)
        addZone(-1, pl, nz(volume[pivLen], 0))

    // Zone state machines
    for i = 0 to array.size(zLvl) - 1
        int side = array.get(zSide, i)
        int st = array.get(zState, i)
        float top = array.get(zTop, i)
        float bot = array.get(zBot, i)
        bool overlaps = low <= top and high >= bot
        bool hit = close >= bot and close <= top and not (close[1] >= bot and close[1] <= top)
        bool pierce = overlaps and not (close >= bot and close <= top)
        // far edge / near edge depend on side: resistance far = top, near = bot
        float far = side == 1 ? top : bot
        float near = side == 1 ? bot : top
        bool bodyBreak = side == 1 ? close > top : close < bot
        bool bodyBreakPrev = side == 1 ? close[1] > top : close[1] < bot
        bool bodyBreakConf = bodyBreak and (breakConfirm == 1 or bodyBreakPrev)
        bool closeOutsideNear = side == 1 ? close < bot : close > top

        if st == ST_ACTIVE
            if hit
                // touch quality (BullByte): false break 1.2 / body 0.8 /
                // consolidation 0.5. (Wick-rejection 1.0 lives on pierces,
                // not hits: a hit means the close is INSIDE the band.)
                float q = 0.8
                if side == 1 and high > top and close <= top
                    q := 1.2
                else if side == -1 and low < bot and close >= bot
                    q := 1.2
                else if open >= bot and open <= top and close >= bot and close <= top
                    q := 0.5
                array.set(zTouches, i, array.get(zTouches, i) + q)
                array.set(zHits, i, array.get(zHits, i) + 1)
                array.set(zState, i, ST_TOUCHED)
                array.set(zMark, i, bar_index)
                if strengthOf(i) * 100 >= strongThr
                    evTouchStrong := true
                refreshLabel(i)
            else if pierce
                array.set(zPierces, i, array.get(zPierces, i) + 1)
                refreshLabel(i)
            // break can also happen without a prior hit on this zone
            if bodyBreakConf
                if showEvents
                    label.new(bar_index, side == 1 ? high : low, "BREAK", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 0), textcolor=color.white, size=size.small)
                evBreak := true
                array.set(zState, i, ST_FLIPPED)
                array.set(zMark, i, bar_index)
                array.set(zSide, i, -side)
                array.set(zFlipped, i, true)
                array.set(zHits, i, 0)
                if array.get(zFirst, i) == 0
                    array.set(zFirst, i, 2)
                box b = array.get(zBox, i)
                box.set_border_color(b, flipCol)
                box.set_bgcolor(b, flipBg)
                refreshLabel(i)

        else if st == ST_TOUCHED
            int markBar = array.get(zMark, i)
            if closeOutsideNear and bar_index - markBar <= 3
                // BOUNCE: close back outside within 3 bars
                if showEvents
                    label.new(bar_index, side == 1 ? top : bot, "BOUNCE", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.green, 0), textcolor=color.white, size=size.small)
                array.set(zBounces, i, array.get(zBounces, i) + 1)
                if array.get(zFirst, i) == 0
                    array.set(zFirst, i, 1)
                array.set(zState, i, ST_ACTIVE)
                array.set(zMark, i, na)
                refreshLabel(i)
            else if bodyBreakConf
                if showEvents
                    label.new(bar_index, side == 1 ? high : low, "BREAK", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 0), textcolor=color.white, size=size.small)
                evBreak := true
                array.set(zState, i, ST_FLIPPED)
                array.set(zMark, i, bar_index)
                array.set(zSide, i, -side)
                array.set(zFlipped, i, true)
                array.set(zHits, i, 0)
                if array.get(zFirst, i) == 0
                    array.set(zFirst, i, 2)
                box b = array.get(zBox, i)
                box.set_border_color(b, flipCol)
                box.set_bgcolor(b, flipBg)
                refreshLabel(i)
            else if bar_index - markBar > 3
                // touch went stale: no bounce, no break — zone stays active
                array.set(zState, i, ST_ACTIVE)
                array.set(zMark, i, na)

        else if st == ST_FLIPPED
            int markBar = array.get(zMark, i)
            if bar_index - markBar > 20
                // break with no retest in 20 bars = violated
                if showEvents
                    label.new(bar_index, side == 1 ? top : bot, "NO RETEST", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.gray, 0), textcolor=color.white, size=size.small)
                array.set(zState, i, ST_DONE_BAD)
                box b = array.get(zBox, i)
                box.set_border_color(b, deadCol)
                box.set_bgcolor(b, deadBg)
                refreshLabel(i)
            else if overlaps
                // retest: price returned into the flipped zone
                if showEvents
                    label.new(bar_index, side == 1 ? bot : top, "RETEST", style=side == 1 ? label.style_label_up : label.style_label_down, color=color.new(color.orange, 0), textcolor=color.white, size=size.small)
                array.set(zState, i, ST_RETEST)
                array.set(zRetestBar, i, bar_index)
                array.set(zHold, i, 0)

        else if st == ST_RETEST
            int rtBar = array.get(zRetestBar, i)
            if bar_index > rtBar
                bool failed = side == 1 ? close < bot : close > top
                if failed
                    if showEvents
                        label.new(bar_index, side == 1 ? high : low, "RETEST FAILED", style=side == 1 ? label.style_label_down : label.style_label_up, color=color.new(color.red, 0), textcolor=color.white, size=size.small)
                    array.set(zState, i, ST_DONE_BAD)
                    box b = array.get(zBox, i)
                    box.set_border_color(b, deadCol)
                    box.set_bgcolor(b, deadBg)
                    refreshLabel(i)
                else
                    array.set(zHold, i, array.get(zHold, i) + 1)
                    if array.get(zHold, i) >= 3
                        if showEvents
                            label.new(bar_index, side == 1 ? bot : top, "HOLD ✓", style=side == 1 ? label.style_label_up : label.style_label_down, color=color.new(color.green, 0), textcolor=color.white, size=size.small)
                        evRetestHeld := true
                        array.set(zState, i, ST_ACTIVE)
                        array.set(zMark, i, na)
                        refreshLabel(i)

    // keep labels glued to the right edge
    for i = 0 to array.size(zLab) - 1
        label.set_xy(array.get(zLab, i), bar_index, array.get(zSide, i) == 1 ? array.get(zTop, i) : array.get(zBot, i))
    for s = 0 to 5
        label ol = array.get(hLab, s)
        if not na(ol)
            label.set_xy(ol, bar_index, s % 2 == 0 ? array.get(hTop, s) : array.get(hBot, s))
